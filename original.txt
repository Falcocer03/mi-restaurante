  <script>
    let tipoComida = "";

    function switchMenu(type, btnElement) {
      document.getElementById('menu-vasco').classList.remove('active');
      document.getElementById('menu-peruano').classList.remove('active');

      const buttons = document.querySelectorAll('.tab-btn');
      buttons.forEach(btn => btn.classList.remove('active-vasco', 'active-peruano'));

      if (type === 'vasco') {
        document.getElementById('menu-vasco').classList.add('active');
        btnElement.classList.add('active-vasco');
      } else {
        document.getElementById('menu-peruano').classList.add('active');
        btnElement.classList.add('active-peruano');
      }
    }

    const fechaInput = document.getElementById('fecha');
    const horaSelect = document.getElementById('hora');
    const noticeBox = document.getElementById('noticeBox');

    const today = new Date().toISOString().split('T')[0];
    fechaInput.setAttribute('min', today);

    fechaInput.addEventListener('change', function() {
      const selectedDate = new Date(this.value + 'T00:00:00');
      const dayOfWeek = selectedDate.getDay();

      horaSelect.innerHTML = '<option value="">-- Selecciona hora --</option>';
      horaSelect.disabled = false;

      if (dayOfWeek === 0) {
        noticeBox.style.display = 'block';
        noticeBox.style.backgroundColor = '#ffe3e3';
        noticeBox.style.color = '#c92a2a';
        noticeBox.innerHTML = '❌ Los domingos cerramos por descanso.';
        horaSelect.disabled = true;
        tipoComida = "";
      } else if (dayOfWeek >= 1 && dayOfWeek <= 5) {
        noticeBox.style.display = 'block';
        noticeBox.style.backgroundColor = '#d3f9d8';
        noticeBox.style.color = '#2b8a3e';
        noticeBox.innerHTML = '🔴 Reserva para Menú Vasco (Horario disponible: 06:00 - 16:30)';
        populateHours(6, 16);
        tipoComida = "Cocina Vasca (Menú del Día)";
      } else if (dayOfWeek === 6) {
        noticeBox.style.display = 'block';
        noticeBox.style.backgroundColor = '#fff3bf';
        noticeBox.style.color = '#e67700';
        noticeBox.innerHTML = '🟡 Reserva para Menú Peruano (Horario disponible: 09:00 - 16:30)';
        populateHours(9, 16);
        tipoComida = "Cocina Peruana (Especial Sábado)";
      }
    });

    function populateHours(startHour, endHour) {
      for (let h = startHour; h <= endHour; h++) {
        let hourFormat = h < 10 ? '0' + h : h;
        let opt1 = document.createElement('option');
        opt1.value = `${hourFormat}:00`;
        opt1.textContent = `${hourFormat}:00`;
        horaSelect.appendChild(opt1);

        if (h !== endHour) {
          let opt2 = document.createElement('option');
          opt2.value = `${hourFormat}:30`;
          opt2.textContent = `${hourFormat}:30`;
          horaSelect.appendChild(opt2);
        }
      }
    }

    document.getElementById('bookingForm').addEventListener('submit', function(e) {
      e.preventDefault();

      const miTelefono = "34614362726";
      const fecha = fechaInput.value;
      const hora = horaSelect.value;
      const personas = document.getElementById('personas').value;
      const nombre = document.getElementById('nombre').value;

      const mensaje = `Hola *AITA INTI*, me gustaría solicitar una reserva:%0A%0A` +
                      `👤 *Nombre:* ${nombre}%0A` +
                      `📅 *Fecha:* ${fecha}%0A` +
                      `⏰ *Hora:* ${hora}%0A` +
                      `👥 *Personas:* ${personas}%0A` +
                      `🍽️ *Experiencia:* ${tipoComida}%0A` +
                      `📍 *Ubicación:* Portuetxe, CP 20018`;

      window.open(`https://wa.me/${miTelefono}?text=${mensaje}`, '_blank');
    });
  </script>
